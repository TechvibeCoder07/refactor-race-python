ORDERS_PROCESSED = []


def calc_subtotal(items):
    subtotal = 0

    for item in items:
        if item["qty"] > 0:
            subtotal += item["price"] * item["qty"]

    return subtotal


def calc_desconto(customer_type, subtotal):
    desconto = 0

    if customer_type == "vip":
        if subtotal >= 1000:
            desconto = subtotal * 0.15
        else:
            desconto = subtotal * 0.10

    elif customer_type == "employee":
        desconto = subtotal * 0.20

    elif customer_type == "regular":
        if subtotal >= 800:
            desconto = subtotal * 0.05

    return desconto


def calc_valor_com_desconto(desconto, subtotal, coupon, customer_type):
    if coupon == "PROMO10":
        desconto += subtotal * 0.10

    if coupon == "PROMO20" and subtotal >= 500:
        desconto += subtotal * 0.20

    if coupon == "VIP50" and customer_type == "vip":
        desconto += 50

    if desconto > subtotal * 0.25:
        desconto = subtotal * 0.25

    return subtotal - desconto


def calc_peso_total(items):
    peso = 0

    for item in items:
        peso += item.get("weight", 0) * item["qty"]

    return peso


def calc_frete(state, peso, subtotal, express):
    if subtotal >= 500 and not express:
        return 0

    if state in {"MG", "SP", "RJ", "ES"}:
        frete = 20 + peso * 0.4
    else:
        frete = 35 + peso * 0.6

    if express:
        frete *= 1.8

    return frete


def calc_imposto(state, valor_com_desconto):
    if state == "MG":
        taxa = 0.07
    elif state == "SP":
        taxa = 0.09
    elif state == "RJ":
        taxa = 0.08
    elif state == "ES":
        taxa = 0.07
    else:
        taxa = 0.12

    return valor_com_desconto * taxa


def calc_pontos(customer_type, valor_com_desconto, frete, imposto):
    if customer_type == "vip":
        divisor = 5
    else:
        divisor = 10

    return int(
        (valor_com_desconto + frete + imposto) / divisor
    )


def busca_duplicados(items):
    duplicados = []

    for i in range(len(items)):
        for j in range(len(items)):
            if i != j:
                if items[i]["name"] == items[j]["name"]:
                    if items[i]["name"] not in duplicados:
                        duplicados.append(items[i]["name"])

    return duplicados


def process_order(customer, items, coupon="", state="MG", express=False):

    customer_type = customer["type"]

    subtotal = calc_subtotal(items)

    desconto = calc_desconto(
        customer_type,
        subtotal
    )

    valor_com_desconto = calc_valor_com_desconto(
        desconto,
        subtotal,
        coupon,
        customer_type
    )

    peso = calc_peso_total(items)

    frete = calc_frete(
        state,
        peso,
        subtotal,
        express
    )

    imposto = calc_imposto(
        state,
        valor_com_desconto
    )

    pontos = calc_pontos(
        customer_type,
        valor_com_desconto,
        frete,
        imposto
    )

    duplicados = busca_duplicados(items)

    total_final = round(
        valor_com_desconto + frete + imposto,
        2
    )

    resultado = {
        "customer": customer["name"],
        "subtotal": round(subtotal, 2),
        "discount": round(desconto, 2),
        "shipping": round(frete, 2),
        "tax": round(imposto, 2),
        "total": total_final,
        "points": pontos,
        "duplicate_products": duplicados
    }

    ORDERS_PROCESSED.append(resultado)

    print("Pedido processado para " + customer["name"])
    print("Subtotal:", subtotal)
    print("Desconto:", desconto)
    print("Frete:", frete)
    print("Imposto:", imposto)
    print("TOTAL:", total_final)

    return resultado
